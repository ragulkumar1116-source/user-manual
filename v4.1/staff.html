Firebase Realtime Database Security Rules
Staff is saved under customers/_staff (allowed by default). To also permit direct root access to staff/ and users/, apply these rules in your Firebase Console.

Go to Firebase Console → aittest-9ec93 → Realtime Database → Rules tab, paste this JSON, and click Publish:

{
  "rules": {
    ".read": false,
    ".write": false,
    "settings": {
      ".read": true,
      ".write": "auth != null"
    },
    "documents": {
      ".read": true,
      ".write": "auth != null",
      ".indexOn": ["title", "category", "updatedAt"]
    },
    "instruments": {
      ".read": true,
      ".write": "auth != null",
      ".indexOn": ["general/name", "general/category", "general/manufacturer", "created_at"]
    },
    "customers": {
      ".read": true,
      ".write": "auth != null",
      "_staff": {
        ".read": "auth != null",
        ".write": "auth != null",
        ".indexOn": ["email", "empId", "role", "status", "updatedAt"]
      },
      "_engineers": {
        ".read": true,
        ".write": "auth != null",
        ".indexOn": ["username", "email", "status", "updatedAt"]
      },
      "_service_reports": {
        ".read": true,
        ".write": "auth != null",
        ".indexOn": ["reportNumber", "companyName", "engineerUsername", "serviceDate", "updatedAt"]
      }
    },
    "staff": {
      ".read": "auth != null",
      ".write": "auth != null",
      ".indexOn": ["email", "empId", "role", "status", "updatedAt"]
    },
    "users": {
